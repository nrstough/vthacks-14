# Production only. deploy.sh rsyncs this file to the box and pip-installs it
# there, so it holds what `backend/app/` needs to run and nothing else.
# Test and experiment tooling lives in requirements-dev.txt.
#
# This list used to also carry pandas, scipy, RapidFuzz, python-multipart and
# httpx2. The first four appeared nowhere in the codebase — not in app/, not in
# tests/, not in experiments/ — and pandas and scipy are the two slowest, most
# fragile installs you could hand a fresh server. httpx2 was wrongly counted with
# them: nothing imports it by name, but starlette's TestClient does, so every
# test module that builds a TestClient needs it. It is a test dependency and
# lives in requirements-dev.txt. The box installs from PyPI with no wheels/ to
# fall back on, so every package here is one more chance for the deploy to die
# halfway through pip install.
#
# pydantic is listed even though fastapi pulls it in anyway: app/schemas.py
# imports it directly, and a direct import that leans on someone else's
# transitive dependency breaks quietly the day that dependency is loosened.
#
# ortools is the one most likely to fail on the box — it needs a wheel built
# for the box's exact Python, which is why Ubuntu 24.04 LTS (Python 3.12) is
# the deliberate choice. If it errors, read the version it names before
# changing anything.
fastapi==0.141.1
# Imported directly by app/forecast/, which runs the frozen forecast
# checkpoints without PyTorch. ortools already pulls numpy in, but the rule
# here is that a direct import is declared: an undeclared one works on this
# laptop and is simply absent on the box.
numpy==2.5.3
ortools==9.15.6755
pydantic==2.13.5
# app/ never imports starlette, but every route runs on it, and fastapi 0.141.1
# asks only for starlette>=0.46.0, so an unpinned install takes whatever is
# newest on PyPI the day the box runs pip. 1.10.0 is the version the suite was
# verified against; move it together with fastapi, on purpose.
starlette==1.10.0
uvicorn==0.53.0
